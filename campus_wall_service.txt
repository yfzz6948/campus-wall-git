[Unit]
Description=Campus Wall Gunicorn Daemon
After=network.target

[Service]
User=root
Group=root
WorkingDirectory=/home/campus-wall
Environment="PATH=/home/campus-wall/.venv/bin"
ExecStart=/bin/bash -c 'export SECRET_KEY="$(python3 -c "import secrets;print(secrets.token_hex(32))")"; exec /home/campus-wall/.venv/bin/gunicorn -w 2 -b 127.0.0.1:5000 --access-logfile /dev/null app:app'
Restart=always

[Install]
WantedBy=multi-user.target
